口径识别：营业收入 命中已有指标「月度营业收入」，定义 SUM(order_amount) - SUM(refund_amount)，已剔除取消单。

生成 SQL 已校验 
SELECT
  DATE_TRUNC('month', o.order_date) AS month,
  SUM(o.order_amount - o.refund_amount) AS revenue,
  LAG(SUM(o.order_amount - o.refund_amount)) OVER (ORDER BY DATE_TRUNC('month', o.order_date)) AS prev_revenue
FROM dwd.fact_orders o
JOIN dim_region r ON o.region_id = r.region_id
WHERE r.region_name = '华东'
  AND o.order_date >= DATE '2024-01-01'
  AND o.order_date <  DATE '2025-01-01'
  AND o.status <> 'cancelled'
GROUP BY 1
ORDER BY 1;

SELECT DATE_TRUNC('month', o.order_date) AS month,
       SUM(o.order_amount - o.refund_amount) AS revenue,
       LAG(SUM(o.order_amount - o.refund_amount))
         OVER (ORDER BY DATE_TRUNC('month', o.order_date)) AS prev_revenue
FROM dwd.fact_orders o
WHERE o.status <> 'cancelled'
GROUP BY 1
ORDER BY 1;
